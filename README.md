Para o Snake funcionar, use este arquivo adicional em:

.github/workflows/snake.yml

name: Generate Snake


on:
  schedule:
    - cron: "0 0 * * *"
  workflow_dispatch:
  push:
    branches:
      - main


jobs:
  generate:
    permissions:
      contents: write


    runs-on: ubuntu-latest


    steps:
      - name: Generate Snake
        uses: Platane/snk@v3
        with:
          github_user_name: luisfpdev
          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark


      - name: Publish Snake
        uses: crazy-max/ghaction-github-pages@v4
        with:
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
